@override
Future<MontyResult> run(
  String code, {
  MontyLimits? limits,
  String? scriptName,
}) async {
  assertNotDisposed('run');
  assertIdle('run');
  markActive();
  try {
    await _ensureInitialized();
    final result = await _bindings.run(
      code,
      limitsJson: _encodeLimits(limits),
      scriptName: scriptName,
    );
    return _translateRunResult(result);
  } finally {
    markIdle();
  }
}
